Mô tả
-----

Có thể dùng lớp này để mở rộng hoặc thay thế triển khai :ref:`MultiplayerAPI<class_MultiplayerAPI>` mặc định thông qua script hoặc extensions.

Ví dụ sau đây mở rộng triển khai mặc định (:ref:`SceneMultiplayer<class_SceneMultiplayer>`) bằng cách ghi nhật ký mọi RPC được thực hiện và mọi đối tượng được cấu hình để replication.

    extends MultiplayerAPIExtension
    class_name LogMultiplayer

    # Chúng ta muốn mở rộng SceneMultiplayer mặc định.
    var base_multiplayer = SceneMultiplayer.new()

    func _init():
        # Chỉ truyền tiếp các tín hiệu cơ sở (được sao chép vào var để tránh tham chiếu vòng)citeturn0search0
        var cts = connected_to_server
        var cf = connection_failed
        var sd = server_disconnected
        var pc = peer_connected
        var pd = peer_disconnected
        base_multiplayer.connected_to_server.connect(func(): cts.emit())
        base_multiplayer.connection_failed.connect(func(): cf.emit())
        base_multiplayer.server_disconnected.connect(func(): sd.emit())
        base_multiplayer.peer_connected.connect(func(id): pc.emit(id))
        base_multiplayer.peer_disconnected.connect(func(id): pd.emit(id))

    func _poll():
        return base_multiplayer.poll()

    # Ghi nhật ký RPC đang được thực hiện và chuyển tiếp nó đến multiplayer mặc định.
    func _rpc(peer: int, object: Object, method: StringName, args: Array) -> Error:
        print("Got RPC for %d: %s::%s(%s)" % [peer, object, method, args])
        return base_multiplayer.rpc(peer, object, method, args)

    # Ghi nhật ký thao tác thêm cấu hình. Ví dụ: đường dẫn gốc (nullptr, NodePath), replication (Node, Spawner|Synchronizer), tùy chỉnh.
    func _object_configuration_add(object, config: Variant) -> Error:
        if config is MultiplayerSynchronizer:
            print("Adding synchronization configuration for %s. Synchronizer: %s" % [object, config])
        elif config is MultiplayerSpawner:
            print("Adding node %s to the spawn list. Spawner: %s" % [object, config])
        return base_multiplayer.object_configuration_add(object, config)

    # Ghi nhật ký thao tác xóa cấu hình. Ví dụ: đường dẫn gốc (nullptr, NodePath), replication (Node, Spawner|Synchronizer), tùy chỉnh.
    func _object_configuration_remove(object, config: Variant) -> Error:
        if config is MultiplayerSynchronizer:
            print("Removing synchronization configuration for %s. Synchronizer: %s" % [object, config])
        elif config is MultiplayerSpawner:
            print("Removing node %s from the spawn list. Spawner: %s" % [object, config])
        return base_multiplayer.object_configuration_remove(object, config)

    # Các hàm này có thể là tùy chọn, nhưng trong trường hợp của chúng ta, ta muốn mở rộng SceneMultiplayer, vì vậy hãy chuyển tiếp mọi thứ.
    func _set_multiplayer_peer(p_peer: MultiplayerPeer):
        base_multiplayer.multiplayer_peer = p_peer
